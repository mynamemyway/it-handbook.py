# String methods

```py
# Поиск и проверка
str.find(sub)           # Возвращает индекс первого вхождения подстроки (или -1)
str.index(sub)          # Возвращает индекс первого вхождения (вызывает ошибку, если не найдено)
str.rfind(sub)          # Возвращает индекс последнего вхождения подстроки
str.count(sub)          # Возвращает количество вхождений подстроки
str.startswith(prefix)  # Проверяет, начинается ли строка с указанного префикса
str.endswith(suffix)    # Проверяется, заканчивается ли строка указанным суффиксом

# Преобразование регистра
str.lower()             # Преобразует все символы в нижний регистр
str.upper()             # Преобразует все символы в верхний регистр
str.capitalize()        # Делает первую букву заглавной, остальные строчными
str.title()             # Делает каждую первую букву слова заглавной
str.swapcase()          # Меняет регистр каждого символа на противоположный
str.casefold()          # Агрессивное приведение к нижнему регистру (для сравнения без учета регистра)

# Выравнивание и заполнение
str.strip(chars)        # Удаляет пробелы или указанные символы с начала и конца
str.lstrip(chars)       # Удаляет пробелы или указанные символы слева
str.rstrip(chars)       # Удаляет пробелы или указанные символы справа
str.center(width)       # Центрирует строку, заполняя пустоты пробелами
str.ljust(width)        # Выравнивает по левому краю, заполняя справа
str.rjust(width)        # Выравнивает по правому краю, заполняя слева
str.zfill(width)        # Заполняет нулями слева до указанной ширины

# Замена и разбиение
str.replace(old, new)   # Заменяет одну подстроку на другую
str.split(sep)          # Разбивает строку на список по разделителю
str.rsplit(sep)         # Разбивает строку на список справа налево
str.splitlines()        # Разбивает строку на список по переносам строк
str.join(iterable)      # Объединяет элементы итерируемого объекта в строку через разделитель

# Проверка содержимого (возвращают bool)
str.isalpha()           # True, если все символы — буквы
str.isdigit()           # True, если все символы — цифры
str.isnumeric()         # True, если все символы — числовые (включая дроби и римские)
str.isalnum()           # True, если строка состоит из букв и/или цифр
str.isspace()           # True, если строка состоит только из пробельных символов
str.islower()           # True, если все буквы в нижнем регистре
str.isupper()           # True, если все буквы в верхнем регистре
str.istitle()           # True, если каждое слово начинается с заглавной буквы

# Форматирование
str.format(*args)       # Форматирует строку, подставляя аргументы в {}
str.encode(encoding)    # Кодирует строку в байты (например, 'utf-8')
str.partition(sep)      # Разбивает строку на кортеж (до, разделитель, после)
