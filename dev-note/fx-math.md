# Math Functions (функции из модуля math)

## Шпаргалка общая `.py`

```py
import math

# Константы (O(1) по памяти и времени)
math.pi             # π ≈ 3.141592... (Число Пи)
math.e              # e ≈ 2.718281... (Число Эйлера)
math.tau            # τ ≈ 6.283185... (Тау = 2*Пи)
math.inf            # Положительная бесконечность (float('inf'))
math.nan            # Значение Not a Number (float('nan'))

# Округление и преобразование (O(1) по памяти и времени)
math.ceil(x)        # ⌈x⌉ Округление до ближайшего большего целого
math.floor(x)       # ⌊x⌉ Округление до ближайшего меньшего целого
math.trunc(x)       # Отбрасывает дробную часть, оставляя только целую
math.modf(x)        # Возвращает кортеж (дробная_часть, целая_часть) в виде float
math.copysign(x, y) # Возвращает x со знаком числа y

# Базовая математика и теория чисел (O(1) по памяти)
math.fabs(x)        # |x| Абсолютное значение (всегда возвращает float)
math.factorial(n)   # n! Факториал числа (Внимание: Time O(N) из-за умножения)
math.gcd(*integers) # НОД — Наибольший общий делитель для группы чисел (Time O(log N))
math.lcm(*integers) # НОК — Наименьшее общее кратное (Time O(log N))
math.isqrt(n)       # Целочисленный квадратный корень (быстрее чем int(n**0.5))
math.perm(n, k)     # Число размещений из n по k (без повторений)
math.comb(n, k)     # Число сочетаний из n по k (биномиальный коэффициент)

# Проверки состояний (O(1) по памяти и времени)
math.isnan(x)       # True, если x является NaN
math.isinf(x)       # True, если x является бесконечностью (inf или -inf)
math.isfinite(x)    # True, если x не является ни NaN, ни бесконечностью
math.isclose(a, b)  # True, если a и b близки друг к другу (с учетом погрешности float)

# Степень и логарифмы (O(1) по памяти и времени)
math.pow(x, y)      # x ** y Возведение в степень (в отличие от **, всегда float)
math.sqrt(x)        # √x Квадратный корень
math.cbrt(x)        # ∛x Кубический корень
math.exp(x)         # e ** x Экспонента
math.log(x, [base]) # Логарифм x по основанию base (по умолчанию натуральный log)
math.log2(x)        # Логарифм по основанию 2 (быстрее и точнее, чем log(x, 2))
math.log10(x)       # Логарифм по основанию 10

# Тригонометрия (аргументы в РАДИАНАХ) (O(1) по памяти и времени)
math.sin(x)         # Синус x
math.cos(x)         # Косинус x
math.tan(x)         # Тангенс x
math.asin(x)        # Арксинус x (возвращает радианы)
math.acos(x)        # Арккосинус x
math.atan(x)        # Арктангенс x
math.atan2(y, x)    # Арктангенс y/x с учетом знака четверти координат

# Перевод углов (O(1) по памяти и времени)
math.degrees(x)     # Конвертирует радианы в градусы
math.radians(x)     # Конвертирует градусы в радианы

# Гиперболические функции (O(1) по памяти и времени)
math.sinh(x)        # Гиперболический синус
math.cosh(x)        # Гиперболический косинус
math.tanh(x)        # Гиперболический тангенс

# Специальные и агрегирующие функции
math.prod(iterable) # Произведение всех элементов (Time O(N), Space O(1))
math.fsum(iterable) # Точная сумма float-чисел без потери точности округления (Time O(N))
math.dist(p, q)     # Евклидово расстояние между двумя точками p и q (кортежи/списки)
math.hypot(*coords) # Гипотенуза Евклидовой нормы: sqrt(sum(x**2 for x in coords))
