# 3. Задачи

1. Отсортировать числа в массиве согласно сумме их цифр.

```py
# Структура данных: хэш-таблица (задача была в ПП)
def sort_by_digit_sum(nums):
    '''
    1. Принимает массив чисел
    2. Возвращает отсортированный по сумме цифр каждого элемента массив
    '''
    # Инициируем пустой словарь
    dict_nums = {}
    # В цикле заполняем словарь key = число, val = сумма цифр числа из массива
    # Используем списочное выражение и строку, чтобы пройти по каждой цифре
    for n in nums:
        # Защищаем от отрицательных чисел по модулю
        dict_nums[n] = sum([int(dig) for dig in str(abs(n))])

    # Инициируем целевой список с отсортированными элементами
    res = []
    # В цикле набираем целевой массив из ключей по отсортированным значениям
    # Используем параметр key, чтобы передать второй элемент кортежа на сорт
    for key, val in sorted(dict_nums.items(), key = lambda x: x[1]):
        res.append(key)

    return res

print(sort_by_digit_sum(nums))

# Расчёт сложности:
# 1. Проход в цикле по каждому числу и перебор цифр это O(n*len(n))
# 2. Timsort работает за O(n*log n), значит берём эту сложность на функцию
```

```py
# 3. Эффективное решение за один проход O(n) с сохранением исходника (тоже из курса ПП)
def two_min_nums(nums):
    '''
    Принимает массив чисел
    Возвращает два минимальных, целых
    '''
    # Задаем положительную бесконечность для сравнения с меньшими
    n_1, n_2 = float('inf'), float('inf')

    # В цикле сравниваем текущее с заданными
    for n in nums:
        # Если число не целое, пропускаем итерацию
        if not float(n).is_integer():
            continue

        if n < n_1:    # если текущее меньше чем первое меньшее
            n_2 = n_1  # сохраняем значение n_2 (как второе меньшшее)
            n_1 = n    # переназначаем n_1

        elif n < n_2:  # если текуще меньше чем второе меньшее
            n_2 = n    # gереназначаем его, n_1 остаётся

    return n_1, n_2

print(two_min_nums(nums))
```

**Простое решения:**

```py
nums = [12, 1, 123, 33, 9, 2.4]

n_1, n_2 = float('inf'), float('inf')

for n in nums:
    # Пропускаем итерацию, если число не целое
    if not float(n).is_integer():
        continue

    # Если n меньше чем n_1, то кладём n_1 в n_2 для след. итерации
    if n < n_1:
        n_2 = n_1
        n_1 = n

    # Если n не меньше чем n_1, но меньше чем n_2, то обновляем n_2
    elif n < n_2:
        n_2 = n

print(n_1, n_2)
```
